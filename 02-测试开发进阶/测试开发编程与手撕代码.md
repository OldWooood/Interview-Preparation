**高频代码题：写一个计时装饰器**

```python
import functools, time

def timer(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        try:
            return func(*args, **kwargs)
        finally:
            cost = time.perf_counter() - start
            print(f"{func.__name__} 耗时 {cost:.4f}s")
    return wrapper

@timer
def slow():
    time.sleep(1)
```

**写一个重试装饰器（测试框架常用）**

```python
import functools, time

def retry(times=3, delay=1, exceptions=(Exception,)):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            last = None
            for i in range(times):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last = e
                    if i < times - 1:
                        time.sleep(delay * (2 ** i))   # 指数退避
            raise last
        return wrapper
    return decorator
```

**高频代码题：用两个线程交替打印 1-100**

```java
public class PrintAbc {
    private static final Object lock = new Object();
    private static int num = 1;
    private static final int MAX = 100;

    public static void main(String[] args) {
        Runnable task = () -> {
            while (true) {
                synchronized (lock) {
                    while (num <= MAX && num % 2 != (Thread.currentThread().getName().equals("奇数") ? 1 : 0)) {
                        try { lock.wait(); } catch (InterruptedException e) { return; }
                    }
                    if (num > MAX) { lock.notifyAll(); return; }
                    System.out.println(Thread.currentThread().getName() + ": " + num++);
                    lock.notifyAll();
                }
            }
        };
        new Thread(task, "奇数").start();
        new Thread(task, "偶数").start();
    }
}
```

    for right, ch in enumerate(s):

            window.remove(s[left])

        window.add(ch)

def find_duplicate(nums):            # 快慢指针（Floyd），O(1) 空间

        slow, fast = nums[slow], nums[nums[fast]]

### 2.2 排序与查找

**快排（必须手写）**

```python
def quick_sort(nums):
    if len(nums) <= 1:
        return nums
    pivot = nums[len(nums) // 2]
    left = [x for x in nums if x < pivot]
    mid = [x for x in nums if x == pivot]
    right = [x for x in nums if x > pivot]
    return quick_sort(left) + mid + quick_sort(right)
```

**原地快排（面试可加分）**

```python
def quick_sort_inplace(nums, low=0, high=None):
    if high is None:
        high = len(nums) - 1
    if low >= high:
        return
    pivot = nums[high]
    i = low
    for j in range(low, high):
        if nums[j] < pivot:
            nums[i], nums[j] = nums[j], nums[i]
            i += 1
    nums[i], nums[high] = nums[high], nums[i]
    quick_sort_inplace(nums, low, i - 1)
    quick_sort_inplace(nums, i + 1, high)
```

**归并排序**

```python
def merge_sort(nums):
    if len(nums) <= 1:
        return nums
    mid = len(nums) // 2
    left, right = merge_sort(nums[:mid]), merge_sort(nums[mid:])
    result, i, j = [], 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    result.extend(left[i:] or right[j:])
    return result
```

**二分查找（含边界）**

```python
def binary_search(nums, target):
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] == target:
            return mid
        elif nums[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1
```

class LRUCache:
    def __init__(self, capacity):
        self.cap = capacity
        self.data = OrderedDict()

    def get(self, key):
        if key not in self.data:
            return -1
        self.data.move_to_end(key)
        return self.data[key]

        if key in self.data:

### 2.5 树

```python
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

def inorder(root):                 # 中序遍历（迭代版）
    result, stack, cur = [], [], root
    while cur or stack:
        while cur:
            stack.append(cur); cur = cur.left
        cur = stack.pop()
        result.append(cur.val)
        cur = cur.right
    return result

def max_depth(root):
    if not root:
        return 0
    return 1 + max(max_depth(root.left), max_depth(root.right))

def is_balanced(root):
    def height(node):
        if not node:
            return 0
        lh, rh = height(node.left), height(node.right)
        if lh == -1 or rh == -1 or abs(lh - rh) > 1:
            return -1
        return 1 + max(lh, rh)
    return height(root) != -1

def level_order(root):             # 层序遍历
    if not root:
        return []
    result, queue = [], [root]
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.pop(0)
            level.append(node.val)
            if node.left: queue.append(node.left)
            if node.right: queue.append(node.right)
        result.append(level)
    return result
```

**限流器（令牌桶简化版）**

```python
import time, threading

class RateLimiter:
    def __init__(self, rate, capacity):
        self.rate, self.capacity = rate, capacity
        self.tokens = capacity
        self.lock = threading.Lock()
        self.last = time.monotonic()

    def acquire(self):
        with self.lock:
            now = time.monotonic()
            self.tokens = min(self.capacity, self.tokens + (now - self.last) * self.rate)
            self.last = now
            if self.tokens >= 1:
                self.tokens -= 1
                return True
            return False
```

### 3.2 JSON 差异对比（Diff 测试核心）

```python
def json_diff(old, new, path="", ignore_keys=()):
    """返回差异列表 [(path, old_value, new_value)]"""
    diffs = []
    if isinstance(old, dict) and isinstance(new, dict):
        for k in set(old) | set(new):
            if k in ignore_keys:
                continue
            p = f"{path}.{k}" if path else k
            if k not in old:
                diffs.append((p, "<missing>", new[k]))
            elif k not in new:
                diffs.append((p, old[k], "<missing>"))
            else:
                diffs.extend(json_diff(old[k], new[k], p, ignore_keys))
    elif isinstance(old, list) and isinstance(new, list):
        if len(old) != len(new):
            diffs.append((path, f"len={len(old)}", f"len={len(new)}"))
        for i, (a, b) in enumerate(zip(old, new)):
            diffs.extend(json_diff(a, b, f"{path}[{i}]", ignore_keys))
    elif old != new:
        diffs.append((path, old, new))
    return diffs
```

## 七、答题策略

1. **先沟通再写**：确认输入输出、边界、复杂度要求；
2. **先暴力再优化**：说清思路，再优化时间和空间；
3. **写测试**：写完举 2–3 个用例（正常/边界/异常），测试思维是加分项；
4. **代码规范**：命名清晰、函数拆分、异常处理；
5. **不会时讲思路**：面试官更看重分析过程，不要沉默。

### 高频考点优先级

```
字符串/数组（40%）> 哈希表（15%）> 链表（10%）
> 排序/二分（10%）> 树（10%）> 动态规划（5%）
> 并发/设计题（10%）
```

## 八、总结

> 测试开发的编程考察 = **基础算法 + 工程代码 + 测试思维**。
> 准备方法：LeetCode 高频 100 题（简单+中等为主）+ 手写装饰器/重试/Diff/断言 + SQL 窗口函数 + Shell 文本处理。
